1. Estructura básica de un programa ▾

Todo programa (sketch) de Arduino tiene dos funciones obligatorias:

estructura-basica.ino 
void setup() {
  // Se ejecuta UNA sola vez, al encender o resetear la placa.
  // Aquí van: pinMode(), Serial.begin(), configuraciones iniciales.
}

void loop() {
  // Se ejecuta en bucle infinito, una y otra vez.
  // Aquí va la lógica principal del programa.
}
  • Cada línea termina en punto y coma ;
  • Los bloques de código van entre llaves { }
  • // inicia un comentario de una línea, /* ... */ un comentario de varias líneas

3. Entradas y salidas digitales ▾
es-digital.ino 
pinMode(pin, MODO);            // INPUT, OUTPUT o INPUT_PULLUP
digitalWrite(pin, HIGH);       // Escribe HIGH o LOW en un pin de salida
int valor = digitalRead(pin);  // Lee HIGH (1) o LOW (0) de un pin de entrada

Ejemplo — encender un LED

led-parpadeante.ino 
void setup() {
  pinMode(8, OUTPUT);
}

void loop() {
  digitalWrite(8, HIGH);
  delay(1000);
  digitalWrite(8, LOW);
  delay(1000);
}
4. Entradas y salidas analógicas (PWM) ▾
es-analogica.ino 
int valor = analogRead(pin);  // Lee un valor entre 0 y 1023 (pines A0-A5)
analogWrite(pin, valor);      // Escribe PWM entre 0 y 255 (pines con ~)

analogRead() convierte un voltaje (0-5V) en un número (0-1023): útil para potenciómetros y sensores. analogWrite() no da un voltaje real, simula uno mediante PWM: útil para controlar brillo o velocidad de motores.

Ejemplo — controlar velocidad con potenciómetro

potenciometro-pwm.ino 
void setup() {
  pinMode(9, OUTPUT);
}

void loop() {
  int lectura = analogRead(A0);            // 0 a 1023
  int pwm = map(lectura, 0, 1023, 0, 255); // Reescala a 0-255
  analogWrite(9, pwm);
}
5. Comunicación serial (monitor serie) ▾
serial.ino 
Serial.begin(9600);           // Inicia comunicación (siempre en setup)
Serial.print("Texto");        // Imprime sin salto de línea
Serial.println("Texto");      // Imprime con salto de línea
Serial.print(variable);       // Imprime el valor de una variable
bool hayDatos = Serial.available(); // >0 si hay datos entrantes
char dato = Serial.read();    // Lee un byte recibido

Ejemplo — depurar valores en el monitor serie

serial-depuracion.ino 
void setup() {
  Serial.begin(9600);
}

void loop() {
  int valor = analogRead(A0);
  Serial.print("Sensor: ");
  Serial.println(valor);
  delay(200);
}
6. Control de tiempo ▾
tiempo.ino 
delay(ms);               // Pausa el programa X milisegundos (BLOQUEANTE)
delayMicroseconds(us);   // Pausa en microsegundos
unsigned long t = millis();  // Milisegundos desde el arranque (no bloquea)
unsigned long t2 = micros(); // Microsegundos desde el arranque

Importante: delay() congela todo el programa. Para hacer varias cosas "al mismo tiempo" (ej. leer un sensor mientras un motor gira) se usa millis():

Ejemplo — temporizador sin bloqueo

millis-temporizador.ino 
unsigned long anterior = 0;
const long intervalo = 1000;

void loop() {
  unsigned long actual = millis();
  if (actual - anterior >= intervalo) {
    anterior = actual;
    // código que se repite cada segundo, SIN bloquear el resto del loop
  }
}

8. Arrays y funciones propias ▾
arrays.ino

int primero = pines[0];           // Acceder (empieza en 0)

Funciones propias

// Función que devuelve un valor
int sumar(int a, int b) {
  return a + b;
}

// Función que no devuelve nada
void parpadear(int pin, int veces) {
  for (int i = 0; i < veces; i++) {
    digitalWrite(pin, HIGH);
    delay(200);
    digitalWrite(pin, LOW);
    delay(200);
  }
}
9. Librerías, interrupciones y motores ▾
librerias.ino 
#include <Servo.h>        // Control de servomotores
#include <Wire.h>         // Comunicación I2C
#include <SPI.h>          // Comunicación SPI
#include <Stepper.h>      // Motores paso a paso (básico)
#include <AccelStepper.h> // Motores paso a paso (avanzado, no bloqueante)

Ejemplo — Servo

servo.ino 
#include <Servo.h>
Servo miServo;

void setup() {
  miServo.attach(9);   // Pin de señal del servo
}

void loop() {
  miServo.write(90);   // Ángulo de 0 a 180 grados
}

Interrupciones

interrupciones.ino 
attachInterrupt(digitalPinToInterrupt(pin), funcion, MODO);
// MODO: LOW, CHANGE, RISING, FALLING

detachInterrupt(digitalPinToInterrupt(pin));

Una interrupción ejecuta una función automáticamente cuando ocurre un evento en un pin (ej. un botón), sin tener que estar revisando en el loop(). En Arduino Uno los pines con interrupción son el 2 y el 3.

Motor paso a paso (stepper) básico

stepper-basico.ino 
#include <Stepper.h>

const int pasosPorVuelta = 200;
Stepper motor(pasosPorVuelta, 8, 9, 10, 11);

void setup() {
  motor.setSpeed(60); // RPM
}

void loop() {
  motor.step(pasosPorVuelta); // Gira una vuelta completa
}
10. Sensores comunes y buenas prácticas ▾
sensores.ino 
// Botón / pulsador
pinMode(2, INPUT_PULLUP);
int estado = digitalRead(2); // LOW = presionado (con pull-up)

// Potenciómetro
int valor = analogRead(A0); // 0-1023

// Sensor ultrasónico HC-SR04
const int trig = 6, echo = 7;
pinMode(trig, OUTPUT);
pinMode(echo, INPUT);
digitalWrite(trig, LOW);
delayMicroseconds(2);
digitalWrite(trig, HIGH);
delayMicroseconds(10);
digitalWrite(trig, LOW);
long duracion = pulseIn(echo, HIGH);
float distanciaCm = duracion * 0.034 / 2;

Buenas prácticas y depuración

  • Usa Serial.println() para revisar valores mientras el programa corre.
  • Nombra tus variables y pines con constantes (const int) en vez de números sueltos ("números mágicos").
  • Evita delay() largos si necesitas que el programa reaccione a sensores o botones al mismo tiempo; usa millis().
  • Verifica siempre las conexiones de GND (tierra) entre Arduino y componentes externos (drivers de motor, fuentes externas, etc.).
  • Si un motor no se mueve, revisa: alimentación externa suficiente, conexiones del driver, y que el pin PWM esté bien elegido (marcado con ~).
  • Comenta tu código explicando el porqué, no solo el qué.
