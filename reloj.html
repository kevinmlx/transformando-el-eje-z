El reloj: impresión 3D + electrónica

El objetivo del proyecto es mostrar cómo se pueden combinar ambas disciplinas. Este reloj es el resultado: un chasis y engranajes impresos en 3D, movidos por un motor paso a paso controlado con Arduino.

Código Arduino

Ver código Arduino (reloj.ino) ▾
reloj.ino 
// ==========================================
// Reloj impulsado por motor paso a paso
// Controlado con Arduino
// ==========================================

// Milisegundos que equivalen a un "minuto" de reloj
// (en la demo se usa 1000 para ver el movimiento rápido;
//  en el reloj real se ajusta a 60000)
#define MILLIS_PER_MIN 1000

// Pasos del motor para una vuelta completa del rotor de minutos
#define STEPS_PER_ROTATION 15360

// Tiempo mínimo de espera entre pasos (afecta la suavidad)
int delaytime = 4;

// Pines de Arduino conectados al driver ULN2003
int port[4] = {5, 4, 3, 2};

// Secuencia de activación de las bobinas del motor:
// cada fila es una fase, cada columna un pin
int seq[4][4] = {
  {  LOW,  LOW, HIGH,  LOW},
  {  LOW,  LOW,  LOW, HIGH},
  { HIGH,  LOW,  LOW,  LOW},
  {  LOW, HIGH,  LOW,  LOW}
};

// Gira el motor "step" pasos (positivo o negativo = sentido)
void rotate(int step) {
  static int phase = 0;
  int i, j;
  // delta 1 = avanzar una fase; delta 3 = retroceder una fase
  int delta = (step > 0) ? 1 : 3;
  int dt = 20;

  step = (step > 0) ? step : -step;
  for(j = 0; j < step; j++) {
    phase = (phase + delta) % 4;
    for(i = 0; i < 4; i++) {
      digitalWrite(port[i], seq[phase][i]);
    }
    delay(dt);
    if(dt > delaytime) dt--;  // acelera poco a poco
  }
  // Corte de energía: apaga todas las bobinas
  // para no calentar el motor ni gastar corriente
  for(i = 0; i < 4; i++) {
    digitalWrite(port[i], LOW);
  }
}

// Configuración inicial
void setup() {
  pinMode(port[0], OUTPUT);
  pinMode(port[1], OUTPUT);
  pinMode(port[2], OUTPUT);
  pinMode(port[3], OUTPUT);
  rotate(-10);  // carrera de aproximación en reversa
  rotate(10);   // toma impulso sin carga pesada
  rotate(STEPS_PER_ROTATION / 60);  // avanza el primer minuto
}

// Bucle principal: cada "minuto" mueve el rotor a su nueva posición
void loop() {
  static long prev_min = 0, prev_pos = 0;
  long min;
  static long pos;

  min = millis() / MILLIS_PER_MIN;
  if(prev_min == min) {
    return;  // todavía no ha pasado un minuto
  }
  prev_min = min;
  // posición absoluta del rotor según los minutos transcurridos
  pos = (STEPS_PER_ROTATION * min) / 60;
  rotate(-10);  // reversa para quitar la tensión de los engranajes
  rotate(10);   // toma impulso para vencer la fricción
  if(pos - prev_pos > 0) {
    rotate(pos - prev_pos);
  }
  prev_pos = pos;
}
